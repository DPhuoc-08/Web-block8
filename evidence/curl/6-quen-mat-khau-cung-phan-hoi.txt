# Quên mật khẩu: email có / không có tài khoản → cùng status, cùng Location, thời gian gần nhau

$ POST /forgot
  Cookie: sid=<đã che · vân tay c0900666>
  _csrf=<đã che>&email=<email nhóm đã che>
→ 303  Location: /forgot?msg=forgot-sent  (204 ms)
  Set-Cookie: sid=<đã che · vân tay c0900666>; Path=/; Expires=Fri, 06 Nov 2026 14:07:46 GMT; HttpOnly; SameSite=Lax
