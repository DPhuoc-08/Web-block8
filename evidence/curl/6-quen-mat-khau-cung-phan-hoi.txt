# Quên mật khẩu: email có / không có tài khoản → cùng status, cùng Location, thời gian gần nhau

  Cookie: sid=<đã che · vân tay de42785b>
  _csrf=<đã che>&email=admin%40example.com

  Set-Cookie: sid=<đã che · vân tay de42785b>; Path=/; Expires=Tue, 03 Nov 2026 06:03:19 GMT; HttpOnly; SameSite=Lax
